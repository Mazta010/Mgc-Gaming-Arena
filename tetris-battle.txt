<script>
const COLS=10,ROWS=20;
const SHAPES={I:[[1,1,1,1]],J:[[1,0,0],[1,1,1]],L:[[0,0,1],[1,1,1]],O:[[1,1],[1,1]],S:[[0,1,1],[1,1,0]],T:[[0,1,0],[1,1,1]],Z:[[1,1,0],[0,1,1]]};
const COLORS={I:'#22d3ee',J:'#3b82f6',L:'#f97316',O:'#facc15',S:'#22c55e',T:'#a78bfa',Z:'#ef4444'};
const KEYS=Object.keys(SHAPES);
let dpr=window.devicePixelRatio||1;
let players=[];
let actx=null;
function beep(f,d,v,t='sine'){try{if(!actx)actx=new(window.AudioContext||window.webkitAudioContext)();let o=actx.createOscillator(),g=actx.createGain();o.type=t;o.frequency.value=f;o.connect(g);g.connect(actx.destination);g.gain.setValueAtTime(v,actx.currentTime);g.gain.exponentialRampToValueAtTime(0.01,actx.currentTime+d);o.start();o.stop(actx.currentTime+d);}catch{}}
function sndAtk(){beep(180,0.2,0.4,'sawtooth');setTimeout(()=>beep(600,0.3,0.4),100)}

function makePlayer(canvas, name){
  let ctx=canvas.getContext('2d');
  function resizeC(){ let r=canvas.getBoundingClientRect(); canvas.width=r.width*dpr; canvas.height=r.width*2*dpr; canvas.style.height=r.width*2+'px'; ctx.setTransform(dpr,0,0,dpr,0,0); }
  resizeC();
  return {canvas,ctx,grid:Array.from({length:ROWS},()=>Array(COLS).fill(0)),cur:null,next:null,hold:null,canHold:true,score:0,lines:0,name,over:false,dropT:0,interval:700, resizeC};
}
function randPiece(){ let k=KEYS[Math.floor(Math.random()*KEYS.length)]; return {k,shape:SHAPES[k].map(r=>[...r]),x:3,y:0,color:COLORS[k]}; }
function rotateM(m){return m[0].map((_,i)=>m.map(r=>r[i]).reverse())}
function collide(g,p,dx=0,dy=0,sh=null){ let s=sh||p.shape; for(let y=0;y<s.length;y++) for(let x=0;x<s[y].length;x++) if(s[y][x]){ let nx=p.x+x+dx, ny=p.y+y+dy; if(nx<0||nx>=COLS||ny>=ROWS|| (ny>=0&&g[ny][nx])) return true; } return false; }

function addGarbage(toPlayer, count){
  if(count<=0) return;
  for(let i=0;i<count;i++){
    let hole=Math.floor(Math.random()*COLS);
    let garbage=Array(COLS).fill('#475569'); garbage[hole]=0;
    toPlayer.grid.shift();
    toPlayer.grid.push(garbage);
  }
  // check if blocked
  if(toPlayer.cur && collide(toPlayer.grid,toPlayer.cur)) toPlayer.over=true;
  showAtk(toPlayer==players[0]?1:0, count);
  sndAtk();
}
function showAtk(pIdx, n){
  let el=document.getElementById(pIdx==0?'atkA':'atkB');
  el.innerText='+'+n+' GARBAGE!';
  el.style.display='block'; el.style.color='#ef4444';
  el.style.animation='none'; el.offsetHeight; el.style.animation='atk.9s forwards';
  setTimeout(()=>el.style.display='none',900);
}

function lock(p, otherIdx){
  p.cur.shape.forEach((r,y)=>r.forEach((v,x)=>{ if(v){ let gy=p.cur.y+y, gx=p.cur.x+x; if(gy>=0) p.grid[gy][gx]=p.cur.color; }}));
  let cleared=0;
  for(let y=ROWS-1;y>=0;y--) if(p.grid[y].every(c=>c)){ p.grid.splice(y,1); p.grid.unshift(Array(COLS).fill(0)); cleared++; y++; }
  if(cleared){
    p.lines+=cleared; p.score+= [0,100,300,600,1000][cleared];
    // BATTLE RULE BOSS: every line cleared = +1 line to enemy!
    let atkCount=cleared; // 1 clear = 1 garbage, kung gusto mo mas brutal gawin cleared*1
    addGarbage(players[otherIdx], atkCount);
  }
  spawn(p);
}
function spawn(p){ p.cur=p.next||randPiece(); p.next=randPiece(); p.canHold=true; if(collide(p.grid,p.cur)){ p.over=true; }}
function drawPlayer(p){
  let sW=p.canvas.getBoundingClientRect().width/COLS;
  let ctx=p.ctx;
  ctx.clearRect(0,0,p.canvas.width,p.canvas.height);
  ctx.strokeStyle='rgba(255,255,255,.04)'; for(let i=0;i<=COLS;i++){ ctx.beginPath(); ctx.moveTo(i*sW,0); ctx.lineTo(i*sW,ROWS*sW); ctx.stroke(); } for(let i=0;i<=ROWS;i++){ ctx.beginPath(); ctx.moveTo(0,i*sW); ctx.lineTo(COLS*sW,i*sW); ctx.stroke(); }
  p.grid.forEach((row,y)=>row.forEach((c,x)=>{ if(c){ ctx.fillStyle=c; ctx.shadowColor=c; ctx.shadowBlur=10; ctx.fillRect(x*sW+1,y*sW+1,sW-2,sW-2); ctx.shadowBlur=0; ctx.fillStyle='rgba(255,255,255,.3)'; ctx.fillRect(x*sW+1,y*sW+1,sW-2,3); }}));
  if(p.cur){ let gy=p.cur.y; while(!collide(p.grid,p.cur,0,gy-p.cur.y+1)) gy++; if(gy!=p.cur.y){ ctx.globalAlpha=0.18; p.cur.shape.forEach((r,yy)=>r.forEach((v,xx)=>{ if(v){ ctx.fillStyle=p.cur.color; ctx.fillRect((p.cur.x+xx)*sW+1,(gy+yy)*sW+1,sW-2,sW-2); }})); ctx.globalAlpha=1; } p.cur.shape.forEach((r,yy)=>r.forEach((v,xx)=>{ if(v){ ctx.fillStyle=p.cur.color; ctx.shadowColor=p.cur.color; ctx.shadowBlur=10; ctx.fillRect((p.cur.x+xx)*sW+1,(p.cur.y+yy)*sW+1,sW-2,sW-2); ctx.shadowBlur=0; }})); }
}
function loop(t){
  if(players.length!=2) return;
  players.forEach((p,i)=>{
    if(p.over) return;
    p.dropT+=16;
    if(p.dropT>p.interval){ if(!collide(p.grid,p.cur,0,1)) p.cur.y++; else lock(p,i==0?1:0); p.dropT=0; }
    drawPlayer(p);
    document.getElementById(i==0?'scoreA':'scoreB').innerText=p.score;
    document.getElementById(i==0?'linesA':'linesB').innerText=p.lines;
  });
  let over=players.filter(p=>p.over);
  if(over.length){
    let winner=players.find(p=>!p.over);
    if(winner || over.length==2){
      let lb=JSON.parse(localStorage.getItem('mgcBattleLB')||'[]');
      if(winner){ lb.push({name:winner.name,score:winner.score,vs:players.find(p=>p!=winner).name,date:Date.now()}); lb.sort((a,b)=>b.score-a.score); lb=lb.slice(0,100); localStorage.setItem('mgcBattleLB',JSON.stringify(lb)); alert('🏆 WINNER: '+winner.name+'! Score '+winner.score); }
      else alert('DRAW!');
      renderLB(); document.getElementById('modal').style.display='flex';
      return;
    }
  }
  requestAnimationFrame(loop);
}
function moveP(idx,d){ let p=players[idx]; if(p.over||!p.cur) return; if(!collide(p.grid,p.cur,d,0)) p.cur.x+=d; }
function rotateP(idx){ let p=players[idx]; if(!p.cur) return; let r=rotateM(p.cur.shape); if(!collide(p.grid,p.cur,0,0,r)) p.cur.shape=r; else if(!collide(p.grid,p.cur,-1,0,r)){p.cur.x--;p.cur.shape=r;} else if(!collide(p.grid,p.cur,1,0,r)){p.cur.x++;p.cur.shape=r;} }
function softP(idx){ let p=players[idx]; if(!collide(p.grid,p.cur,0,1)) p.cur.y++; }
function hardP(idx){ let p=players[idx]; let d=0; while(!collide(p.grid,p.cur,0,d+1)) d++; p.cur.y+=d; lock(p, idx==0?1:0); }
function holdP(idx){ let p=players[idx]; if(!p.canHold) return; if(!p.hold){ p.hold=p.cur; spawn(p);} else { let tmp=p.hold; p.hold=p.cur; p.cur=tmp; p.cur.x=3; p.cur.y=0; } p.canHold=false; }

function startBattle(){
  let a=document.getElementById('inA').value||'Mazta'; let b=document.getElementById('inB').value||'Kalaban';
  localStorage.setItem('mgcBattleA',a); localStorage.setItem('mgcBattleB',b);
  document.getElementById('nameA').innerText=a; document.getElementById('nameB').innerText=b;
  let ca=document.getElementById('boardA'), cb=document.getElementById('boardB');
  players=[makePlayer(ca,a), makePlayer(cb,b)];
  players.forEach(p=>{ p.next=randPiece(); spawn(p); });
  document.getElementById('modal').style.display='none';
  actx=new (window.AudioContext||window.webkitAudioContext)();
  window.addEventListener('resize',()=>players.forEach(p=>p.resizeC()));
  requestAnimationFrame(loop);
}
function renderLB(){ let lb=JSON.parse(localStorage.getItem('mgcBattleLB')||'[]'); let h=''; lb.forEach((e,i)=>{ h+=`<div class="lb-row"><span>${i+1}. ${e.name} vs ${e.vs}</span><span>${e.score}</span></div>`}); if(!lb.length) h='<div style="opacity:.5;font-size:11px">Wala pa battle winner - ikaw una boss!</div>'; document.getElementById('lbList').innerHTML=h; }
renderLB();
document.addEventListener('keydown',e=>{
  if(document.getElementById('modal').style.display!='none') return;
  if(e.key=='a'||e.key=='A') moveP(0,-1);
  if(e.key=='d'||e.key=='D') moveP(0,1);
  if(e.key=='s'||e.key=='S') softP(0);
  if(e.key=='q'||e.key=='Q'||e.key=='e'||e.key=='E') rotateP(0);
  if(e.key=='f'||e.key=='F') holdP(0);
  if(e.key=='g'||e.key=='G') hardP(0);
  if(e.key=='ArrowLeft') { e.preventDefault(); moveP(1,-1); }
  if(e.key=='ArrowRight') { e.preventDefault(); moveP(1,1); }
  if(e.key=='ArrowDown') { e.preventDefault(); softP(1); }
  if(e.key=='ArrowUp'||e.key=='x'||e.key=='X') rotateP(1);
  if(e.code=='Space') { e.preventDefault(); hardP(1); }
  if(e.key=='c'||e.key=='C') holdP(1);
});
document.getElementById('inA').value=localStorage.getItem('mgcBattleA')||'Mazta';
document.getElementById('inB').value=localStorage.getItem('mgcBattleB')||'Kalaban';
</script>
